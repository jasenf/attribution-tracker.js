First-touch attribution with Next.js App Router

Attribution Tracker project guide. Adapted with AI assistance from the project's integration example and reviewed against the linked platform documentation. Verify your own consent-manager and form configuration before deployment.

To run the complete example, download or clone the example source folder, change into examples/nextjs-app-router, and run the commands below. In an existing Next.js app, install @feech/attribution-tracker and adapt the component and action shown on this page. The Server Action is a demonstration of receiving data, not production email validation or a CRM write.

This example captures selected first-touch values in a Client Component and sends them to a Server Action as hidden form fields. It targets Next.js 16.3.6 and React 19.3.0.

npm install
npm run dev

Open http://localhost:3000/?utm_source=newsletter&utm_campaign=launch, grant the demo consent, and submit the form. The result shows the attribution fields received by the Server Action.

The consent buttons are a test control, not a consent manager. In a real application, replace the local consent state with your consent management platform's current decision and change callback. Decide whether the tracker belongs under your analytics, marketing, or another storage category based on your own use and policy.

The tracker runs inside app/lead-form.js because it needs browser APIs. Next.js pages and layouts are Server Components by default. The Server Action allowlists four attribution values and trims their length before returning them. Replace the example return value with your validated CRM or database write.

The example checks Global Privacy Control and Do Not Track even after its demo consent is granted. It also clears the stored first touch when consent is withdrawn. It intentionally does not send the full landing page or referrer, since URLs can contain sensitive query data.

References: Next.js Client Components, Next.js forms and Server Actions, and React useActionState.

app/lead-form.js

'use client';

import { useActionState, useEffect, useRef, useState } from 'react';
import AttributionTracker from '@feech/attribution-tracker';
import { submitLead } from './actions';

const initialResult = { message: '', attribution: {} };
const allowedFields = ['utm_source', 'utm_medium', 'utm_campaign', 'ref'];

function privacySignalBlocksStorage() {
  return navigator.globalPrivacyControl === true || navigator.doNotTrack === '1';
}

function selectAttribution(data) {
  return Object.fromEntries(
    allowedFields
      .filter((key) => data?.[key])
      .map((key) => [key, data[key]])
  );
}

export default function LeadForm() {
  const [result, formAction, pending] = useActionState(submitLead, initialResult);
  const [consent, setConsent] = useState(false);
  const [blocked, setBlocked] = useState(false);
  const [attribution, setAttribution] = useState({});
  const trackerRef = useRef(null);

  useEffect(() => {
    const privacyBlocked = privacySignalBlocksStorage();
    setBlocked(privacyBlocked);

    if (!consent || privacyBlocked) return;

    const tracker = new AttributionTracker({
      consent: () => consent && !privacySignalBlocksStorage()
    });

    trackerRef.current = tracker;
    setAttribution(selectAttribution(tracker.getAll()));
  }, [consent]);

  function withdrawConsent() {
    trackerRef.current?.clear();
    trackerRef.current = null;
    setAttribution({});
    setConsent(false);
  }

  return (
    <section>
      <div className="consent-row">
        <button
          type="button"
          onClick={() => setConsent(true)}
          disabled={consent || blocked}
        >
          Grant demo consent
        </button>
        <button type="button" onClick={withdrawConsent} disabled={!consent}>
          Withdraw consent and clear
        </button>
      </div>

      <p className="status" aria-live="polite">
        {blocked
          ? 'A browser privacy signal blocks attribution storage.'
          : consent
            ? `Stored fields: ${Object.keys(attribution).join(', ') || 'none on this URL'}`
            : 'Attribution storage is off until consent is granted.'}
      </p>

      <form action={formAction}>
        <label htmlFor="email">Email</label>
        <input id="email" name="email" type="email" defaultValue="reader@example.com" required />

        {Object.entries(attribution).map(([key, value]) => (
          <input key={key} type="hidden" name={`attribution_${key}`} value={value} />
        ))}

        <button type="submit" disabled={pending}>
          {pending ? 'Sending…' : 'Send test lead'}
        </button>
      </form>

      {result.message && (
        <div className="result" aria-live="polite">
          <p>{result.message}</p>
          <pre>{JSON.stringify(result.attribution, null, 2)}</pre>
        </div>
      )}
    </section>
  );
}

app/actions.js

'use server';

const attributionFields = [
  'attribution_utm_source',
  'attribution_utm_medium',
  'attribution_utm_campaign',
  'attribution_ref'
];

export async function submitLead(previousState, formData) {
  const email = String(formData.get('email') || '').trim();
  const attribution = {};

  for (const name of attributionFields) {
    const value = formData.get(name);
    if (typeof value === 'string' && value) {
      attribution[name.replace('attribution_', '')] = value.slice(0, 500);
    }
  }

  // Replace this return value with a validated write to your CRM or database.
  return {
    message: email ? `Received a test lead for ${email}.` : 'Email is required.',
    attribution
  };
}
